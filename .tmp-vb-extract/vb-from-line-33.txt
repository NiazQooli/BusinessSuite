Private Sub cmdUncheck_Click()
On Error GoTo ErrorHandler

   Dim blnTransAppCnn   As Boolean
   Dim blnTransAppCnnME As Boolean
   
   If Len(Trim(mebSalesEntryID.text)) = 0 Then
      MsgBox "Sales Invoice No. must be entered properly.", vbExclamation + vbOKOnly, App.ProductName
      Exit Sub
   End If
   
   If UnCheckMsg() = vbYes Then
   
      Screen.MousePointer = vbHourglass
      
      AppCnn.BeginTrans
      blnTransAppCnn = True
      
         Call UnCheckSalesInvoice(AppCnn, "BG", mebSalesEntryID.text)
         
         Call AddActivity(AppCnn, mstrFrmCaption, enzActivityUnCheck, "BG - " & Right("00000" & mebSalesEntryID.text, 6), gstrMachineName, _
                          Val(gstrOrgID))

      AppCnn.CommitTrans
      blnTransAppCnn = False
      
      Call ClearFormObjects(Me)
      Call fpgLockGrid(fpgTransaction, True, True)
      Call enztb.SetToolbarNewVersion(enzButtonLoad, mUserPriv)
      Call DisplayDocStatusNewVersion(False, False, True)

      Screen.MousePointer = vbDefault

   End If

   Exit Sub

ErrorHandler:
   Screen.MousePointer = vbDefault

   If blnTransAppCnn = True Then
      AppCnn.RollbackTrans
      blnTransAppCnn = False
   End If
   
   MsgBox "Error Code: " & Err.Number & vbCrLf & _
          "Error Description: " & Err.Description, vbCritical + vbOKOnly, App.ProductName
End Sub

Private Sub ExportReport(pstrExportOption As String, pstrFileName As String, prptReport As ActiveReport)
   Dim oPDF As New ARExportPDF
   
   Dim objFSO  As New FileSystemObject
   
   If objFSO.FolderExists(App.Path & "\_Sales Invoices PDF") = False Then
      objFSO.CreateFolder (App.Path & "\_Sales Invoices PDF")
   End If
   
   oPDF.Filename = App.Path & "\_Sales Invoices PDF\" & pstrFileName & ".pdf"
   
   prptReport.Export oPDF
End Sub

Private Sub Form_KeyDown(KeyCode As Integer, Shift As Integer)
   If KeyCode = vbKeyN And Shift = 2 And enztb.AddNew = True Then
      Call enztb_ClickNew
         
   ElseIf KeyCode = vbKeyE And Shift = 2 And enztb.Edit = True Then
      Call enztb_ClickEdit
      
   ElseIf KeyCode = vbKeyDelete And Shift = 2 And enztb.Delete = True Then
      Call enztb_ClickDelete
      
   ElseIf KeyCode = vbKeyK And Shift = 2 And enztb.Check = True Then
      Call enztb_ClickCheck
      
   ElseIf KeyCode = vbKeyO And Shift = 2 And enztb.Post = True Then
      Call enztb_ClickPost
      
   ElseIf KeyCode = vbKeyF3 And enztb.Find = True Then
      Call enztb_ClickFind
      
   ElseIf KeyCode = vbKeyF12 And enztb.Save = True Then
      Call enztb_ClickSave
      
   End If
End Sub

   End If
End Sub

Private Sub enztb_ClickDelete()
On Error GoTo ErrorHandle

   Dim blnTransAppCnn   As Boolean

   mvarFormMode = enzDeleteMode
   Call SetFormMode

   If DeleteMsg() = vbYes Then

      Screen.MousePointer = vbHourglass

      AppCnn.BeginTrans
      blnTransAppCnn = True

         AppCnn.Execute "DELETE FROM " & mstrSalesTableName & "Detail WHERE CompanyID = " & Val(gstrOrgID) & " AND " & _
                        "FinYear = '" & gstrFinancialYear & "' AND BillType = 'SI' AND " & _
                        "SalesEntryType = 'BG' AND SalesEntryID = " & mebSalesEntryID.text

         AppCnn.Execute "DELETE FROM " & mstrSalesTableName & " WHERE CompanyID = " & Val(gstrOrgID) & " AND " & _
                        "FinYear = '" & gstrFinancialYear & "' AND BillType = 'SI' AND " & _
                        "SalesEntryType = 'BG' AND SalesEntryID = " & mebSalesEntryID.text
                        
         Call AddActivity(AppCnn, mstrFrmCaption, enzActivityDelete, "BG - " & Right("00000" & mebSalesEntryID.text, 6), gstrMachineName, _
                          Val(gstrOrgID))

      AppCnn.CommitTrans
      blnTransAppCnn = False

      Screen.MousePointer = vbDefault

   End If

   Call ClearFormObjects(Me)

   With fpgTransaction
      .ClearRange 1, 1, .MaxCols, .MaxRows, True
      .MaxRows = 2
   End With

   Call enztb.SetToolbarNewVersion(enzButtonLoad, mUserPriv)
   Call DisplayDocStatusNewVersion(False, False, True)

   mvarFormMode = enzViewMode
   Call SetFormMode

   Exit Sub

ErrorHandle:
   If blnTransAppCnn = True Then
      AppCnn.RollbackTrans
      blnTransAppCnn = False
   End If

   Select Case Err.Number

      Case -2147217873
         MsgBox "System is unable to perform this action becuase related record exist in databae.", _
                vbCritical + vbOKOnly, App.ProductName

      Case Else
         MsgBox "Error Code: " & Err.Number & vbCrLf & _
                "Error Description: " & Err.Description, vbCritical + vbOKOnly, App.ProductName

   End Select
End Sub

Private Sub enztb_ClickSave()
   If IsDate(mebSalesEntryDate.text) = False Then
      MsgBox "Sales Invoice Date is not valid.", vbExclamation + vbOKOnly, App.ProductName
      mebSalesEntryDate.SetFocus
      Exit Sub

   End If

'''      Exit Sub
'''   End If

      Exit Sub
   End If

   If ApplySavingRoutine(mvarFormMode) = False Then Exit Sub

   mvarFormMode = enzViewMode
   Call SetFormMode

   Call ClearFormObjects(Me)
   Call SetObjectsState(Me, False)
   Call fpgLockGrid(fpgTransaction, True, True)
   Call enztb.SetToolbarNewVersion(enzButtonLoad, mUserPriv)
   Call DisplayDocStatusNewVersion(False, False, True)

Private Sub fpgTransaction_Advance(ByVal AdvanceNext As Boolean)
   Dim varStr As Variant

   If fpgTransaction.ActiveRow = fpgTransaction.MaxRows Then

      With fpgTransaction
      
         .GetText 1, .ActiveRow, varStr
         If Len(varStr) <= 0 Then Exit Sub

         .GetText 4, .ActiveRow, varStr
         If Val(varStr) <= 0 Then Exit Sub
         
         .GetText 7, .ActiveRow, varStr
         If Val(varStr) <= 0 Then Exit Sub

         .MaxRows = .MaxRows + 1
         .SetActiveCell 1, fpgTransaction.MaxRows
         
         .Col = -1
         .Row = -1
         .BackColor = RGB(255, 255, 255)
         .SetCellBorder -1, -1, -1, -1, SS_BORDER_TYPE_NONE, RGB(255, 0, 0), CellBorderStyleSolid
      
         .Col = -1
         .Row = .ActiveRow
         .BackColor = RGB(192, 192, 192)
         .SetCellBorder -1, .ActiveRow, -1, .ActiveRow, SS_BORDER_TYPE_OUTLINE, RGB(255, 0, 0), CellBorderStyleSolid
         
      End With

   End If
End Sub

Private Function ValidateFPGrid() As Boolean
   Dim i                As Long
   Dim lngMaxRows       As Long

   Dim varFG_ID         As Variant
   Dim varFGName        As Variant
   Dim varNoOfPacks     As Variant
   Dim varBasicRate     As Variant
   Dim varVATPersentage As Variant

   ValidateFPGrid = False

   lngMaxRows = fpgTransaction.MaxRows
   i = 1

   Do While i <= lngMaxRows

      With fpgTransaction

         .GetText 1, i, varFG_ID
         .GetText 3, i, varFGName
         .GetText 4, i, varNoOfPacks
         .GetText 7, i, varBasicRate
         .GetText 25, i, varVATPersentage

         If Len(varFG_ID) = 0 And Val(varNoOfPacks) = 0 And Val(varBasicRate) = 0 Then

            If fpgTransaction.MaxRows > 1 Then
               .Row = i
               .Action = 5
               .MaxRows = .MaxRows - 1
               lngMaxRows = lngMaxRows - 1

               i = i - 1

            Else
               MsgBox "At least one transaction is required in detail section.", vbExclamation + vbOKOnly, App.ProductName
               fpgTransaction.SetFocus
               Exit Function

            End If

         Else
            If Len(varFG_ID) <> Len(gsobstrFGIDMask) Or Len(varFGName) = 0 Then
               MsgBox "FG Code is not valid." & vbCrLf & "Line No.: " & i & ".", vbExclamation + vbOKOnly, App.ProductName
               fpgTransaction.SetFocus
               fpgTransaction.SetActiveCell 1, i
               Exit Function

            Else
               If GetIDTitle("tblInvItemFG", "FG_ID", CStr(varFG_ID), "FG_ID") = "" Then
                  MsgBox "FG Code does not exist." & vbCrLf & "Line No.: " & i & ".", vbExclamation + vbOKOnly, App.ProductName
                  fpgTransaction.SetFocus
                  fpgTransaction.SetActiveCell 1, i
                  Exit Function
               End If
               
            End If
               
            If Val(varNoOfPacks) = 0 Then
               MsgBox "No. Of Packs cannot be zero(0)." & vbCrLf & "Line No.: " & i & ".", vbExclamation + vbOKOnly, App.ProductName
               fpgTransaction.SetFocus
               fpgTransaction.SetActiveCell 4, i
               Exit Function
            End If
            
            If Val(varBasicRate) = 0 Then
               MsgBox "Basic Rate cannot be zero(0)." & vbCrLf & "Line No.: " & i & ".", vbExclamation + vbOKOnly, App.ProductName
               fpgTransaction.SetFocus
               fpgTransaction.SetActiveCell 7, i
               Exit Function
            End If
            
            If Val(varVATPersentage) > 0 And Val(varVATPersentage) <> 20 Then
               MsgBox "Standard VAT Rate is 20%. Please check input." & vbCrLf & "Line No.: " & i & ".", vbExclamation + vbOKOnly, App.ProductName
               fpgTransaction.SetFocus
               fpgTransaction.SetActiveCell 25, i
               Exit Function
            End If

         End If

      End With

      i = i + 1

   Loop

   ValidateFPGrid = True
End Function

Private Sub CalculateTotals()
   Dim intCounter                      As Integer
   
   Dim dblTotalWeight                  As Double, dblTotalTons          As Double
   Dim dblTotalExcSTValue              As Double, dblTotalSTValue       As Double
   Dim dblTotalIncSTValue              As Double, dblTotalEmptyCartons  As Double
   Dim dblTotalFreightCharges          As Double, dblTotalSoapCharges   As Double
   Dim dblTotalSEDValue                As Double
   
   Dim dblTotalSTValue_17              As Double
   Dim dblTotalSTValue_03              As Double
   
   Dim dblTotalDisc1_Percentage_Value  As Variant
   Dim dblTotalDisc2_PerPack_Value     As Variant
   
   Dim dblTotalVATValue                As Variant

   Dim varNoOfPacks                    As Variant, varWeight            As Variant
   Dim varTons                         As Variant, varExcSTValue        As Variant
   Dim varSTValue                      As Variant, varIncSTValue        As Variant
   Dim varEmptyCartons                 As Variant, varForTonsConv       As Variant
   Dim varFreightCharges               As Variant, varSoapCharges       As Variant
   Dim varSEDValue                     As Variant
   
   Dim varSTValue_17                   As Variant
   Dim varSTValue_03                   As Variant
   
   Dim varDisc1_Percentage_Value       As Variant
   Dim varDisc2_PerPack_Value          As Variant
   
   Dim varVATValue                     As Variant

   dblTotalWeight = 0
   dblTotalTons = 0
   dblTotalExcSTValue = 0
   dblTotalSEDValue = 0
   dblTotalSTValue = 0
   
   dblTotalSTValue_17 = 0
   dblTotalSTValue_03 = 0
   
   dblTotalIncSTValue = 0
   dblTotalEmptyCartons = 0
   dblTotalFreightCharges = 0
   dblTotalSoapCharges = 0
   
   dblTotalDisc1_Percentage_Value = 0
   dblTotalDisc2_PerPack_Value = 0
   
   dblTotalVATValue = 0

   fpgTransaction.ReDraw = False
   
   With fpgTransaction
      infTotalWeight.Caption = Format(0#, "##,##0.00")

      For intCounter = 1 To .MaxRows
         .GetText 5, intCounter, varTons
         .GetText 18, intCounter, varForTonsConv
         dblTotalTons = Val(dblTotalTons) + ((Val(Format(varTons, "####0.00")) * Val(varForTonsConv)) / 1000)
      Next

      infTotalWeight.Caption = Format(Val(dblTotalTons), "##,##0.00")
   End With
   
   With fpgTransaction
      infExcSEDnSTValue.Caption = Format(0#, "##,##0.00")

      For intCounter = 1 To .MaxRows
         .GetText 13, intCounter, varExcSTValue
         dblTotalExcSTValue = Val(dblTotalExcSTValue) + Val(Format(varExcSTValue, "####0.00"))
      Next

      infExcSEDnSTValue.Caption = Format(Val(dblTotalExcSTValue), "##,##0.00")
   End With
   
   With fpgTransaction
      infVATValue.Caption = Format(0#, "##,##0.00")

      For intCounter = 1 To .MaxRows
         .GetText 26, intCounter, varVATValue
         dblTotalVATValue = Val(dblTotalVATValue) + Val(Format(varVATValue, "####0.00"))
      Next

      infVATValue.Caption = Format(Val(dblTotalVATValue), "##,##0.00")
   End With
   
   infNetValue.Caption = Format(Val(dblTotalExcSTValue) + Val(dblTotalVATValue), "##,##0.00")
   
   fpgTransaction.ReDraw = True
End Sub

Private Function ApplySavingRoutine(pFormMode As enzFormMode) As Boolean
On Error GoTo ErrorHandler

   Dim blnTransAppCnn   As Boolean
   
   ApplySavingRoutine = True

   If pFormMode = enzAddMode Then

      mebSalesEntryID.text = NEW_GetNewSalesEntryID_Typewise("BG")

   End If

   AppCnn.BeginTrans
   blnTransAppCnn = True

      Call SaveRecord(AppCnn, pFormMode)
      
      Call AddActivity(AppCnn, mstrFrmCaption, IIf(mvarFormMode = enzAddMode, enzActivityAdd, enzActivityEdit), _
                       "BG - " & Right("00000" & mebSalesEntryID.text, 6), gstrMachineName, Val(gstrOrgID))

   AppCnn.CommitTrans
   blnTransAppCnn = False
   
   Exit Function

ErrorHandler:
   If blnTransAppCnn = True Then
      AppCnn.RollbackTrans
      blnTransAppCnn = False
   End If
   
   ApplySavingRoutine = False

   MsgBox "Error Code: " & Err.Number & vbCrLf & _
          "Error Description: " & Err.Description, vbCritical + vbOKOnly, App.ProductName
End Function

Private Function CheckEntryDate(pDateObject As Control, pstrType As String) As Boolean

   rsTemp.Open "SELECT MAX(SalesEntryDate) AS MaxDate FROM " & mstrSalesTableName & " WHERE " & _

      If IsNull(rsTemp.Fields("MaxDate").Value) = False Then
         If CDate(Format(pDateObject.text, gstrDateFormat)) < _
            CDate(Format(rsTemp.Fields("MaxDate").Value, gstrDateFormat)) Then

            pDateObject.SetFocus

         End If
      End If
      
   End If

   CheckEntryDate = True

   rsTemp.Close
   Set rsTemp = Nothing
End Function

Private Sub GridRowCalculations(plngRowNo As Long, Optional pdblSTPercentage As Double, Optional pdblSEDPercentage As Double)
   Dim varFG_ID                  As Variant
   Dim varNoOfPacks              As Variant, varWeight            As Variant, varBasicRate      As Variant
   Dim varSpecialDiscount        As Variant, varExcSEDnSTRate     As Variant, varIncSEDnSTValue As Variant
   Dim varSTRate                 As Variant, varSTAmount          As Variant, varExcSEDnSTValue As Variant
   Dim varEmptyCartons           As Variant, varNoOfUnitsInPack   As Variant, varUnitWeight     As Variant
   Dim varEmptyCartonPercentage  As Variant, varSEDRate           As Variant, varSEDAmount      As Variant
   
   Dim varSTAmount_Reg           As Variant
   Dim varSTAmount_Ext           As Variant
   
   Dim varDisc1_Percentage       As Variant
   Dim varDisc1_Percentage_Value As Variant
   Dim varDisc2_PerPack          As Variant
   Dim varDisc2_PerPack_Value    As Variant
   
   Dim varNetRate                As Variant
   
   Dim varNetRate1               As Variant
   Dim varNetRate2               As Variant
   
   Dim varIncSEDValue            As Variant
   Dim varIncSTValue             As Variant
   
   'FG CODE
   fpgTransaction.GetText 1, plngRowNo, varFG_ID
   
   'NO. OF PACKS
   fpgTransaction.GetText 4, plngRowNo, varNoOfPacks
   
   'BASIC RATE AND SPECIAL DISC.
   fpgTransaction.GetText 7, plngRowNo, varBasicRate
   fpgTransaction.GetText 8, plngRowNo, varSpecialDiscount
   
   'S.E.D RATE
   fpgTransaction.GetText 25, plngRowNo, varSEDRate
   
   If pdblSTPercentage <> 0 Then
      fpgTransaction.SetText 11, plngRowNo, pdblSTPercentage
      
      varSTRate = pdblSTPercentage
      
   Else
      'S.T RATE
      fpgTransaction.GetText 11, plngRowNo, varSTRate
      
   End If
         
   If pdblSEDPercentage <> 0 Then
      fpgTransaction.SetText 25, plngRowNo, pdblSEDPercentage
      
      varSEDRate = pdblSEDPercentage
      
   Else
      'S.E.D RATE
      fpgTransaction.GetText 25, plngRowNo, varSEDRate
      
   End If
      
   'DISCOUNT 2 - PER PACK
   fpgTransaction.GetText 29, plngRowNo, varDisc2_PerPack
   
   'UNIT WEIGHT AND NO. OF UNITS IN PACK
   fpgTransaction.GetText 16, plngRowNo, varNoOfUnitsInPack
   fpgTransaction.GetText 17, plngRowNo, varUnitWeight
   
   'WEIGHT
   Dim dblWeight As Double
   dblWeight = Val(varNoOfUnitsInPack) * Val(varUnitWeight) * Val(varNoOfPacks)
   fpgTransaction.SetText 5, plngRowNo, Val(dblWeight)
   
   'EXC. S.E.D AND S.T RATE
   If Val(varSTRate) > 0 Then
      fpgTransaction.SetText 9, plngRowNo, _
                                Format(Val(varBasicRate) - Val(Format((Val(varSpecialDiscount) / (1 + Val(varSTRate) / 100)), _
                                "####0.000")), "##,##0.000")
   Else
      fpgTransaction.SetText 9, plngRowNo, Format(Val(varBasicRate) - _
                                                  Val(Format((Val(varSpecialDiscount)), "####0.000")), "##,##0.000")
   End If
   fpgTransaction.GetText 9, plngRowNo, varExcSEDnSTRate
   varExcSEDnSTRate = Format(varExcSEDnSTRate, "####0.000")
   
   'EXC. S.E.D AND S.T VALUE
   fpgTransaction.SetText 13, plngRowNo, Format(Val(varNoOfPacks) * Val(varExcSEDnSTRate), "##,##0.00")
   fpgTransaction.GetText 13, plngRowNo, varExcSEDnSTValue
   
   'NET RATE
   fpgTransaction.SetText 14, plngRowNo, Val(varExcSEDnSTRate) + _
                                         Val(Format(Val(varExcSEDnSTRate) * (Val(varSEDRate) / 100), "####0.000")) + _
                                         Val(Format(Val(varExcSEDnSTRate) * (Val(varSTRate) / 100), "####0.000"))
   fpgTransaction.GetText 14, plngRowNo, varNetRate
   
   varNetRate1 = Val(varExcSEDnSTRate) + Val(Format(Val(varExcSEDnSTRate) * (Val(varSEDRate) / 100), "####0.000"))
   varNetRate2 = Val(varExcSEDnSTRate) + Val(Format(Val(varExcSEDnSTRate) * (Val(varSTRate) / 100), "####0.000"))
      
   varIncSEDValue = Format(Val(varNoOfPacks) * Val(Format(varNetRate1, "####0.000")), "##,##0.00")
   varIncSTValue = Format(Val(varNoOfPacks) * Val(Format(varNetRate2, "####0.000")), "##,##0.00")
   
   'INC. S.E.D AND S.T VALUE
   fpgTransaction.SetText 10, plngRowNo, Format(Val(varNoOfPacks) * Val(Format(varNetRate, "####0.000")), "##,##0.00")
   fpgTransaction.GetText 10, plngRowNo, varIncSEDnSTValue
   
   If Val(varSEDRate) > 0 Then
      varSEDAmount = Format(Val(Format(varIncSEDValue, "####0.00")) - Val(Format(varExcSEDnSTValue, "####0.00")), "##,##0.00")
   
      'S.E.D VALUE
      fpgTransaction.SetText 26, plngRowNo, varSEDAmount
      
   Else
      'S.E.D VALUE
      fpgTransaction.SetText 26, plngRowNo, 0
   
   End If
   
   varSTAmount_Reg = 0#
   varSTAmount_Ext = 0#
   
   fpgTransaction.SetText 32, plngRowNo, 0#
   fpgTransaction.SetText 33, plngRowNo, 0#
   
   If Val(varSTRate) > 0 Then
      varSTAmount = Format(Val(Format(varIncSTValue, "####0.00")) - Val(Format(varExcSEDnSTValue, "####0.00")), "##,##0.00")
   
      'S.T VALUE
      fpgTransaction.SetText 12, plngRowNo, varSTAmount
      
      'CALCULATION OF EXTRA S.TAX
      If pdblSTPercentage = 20 Or pdblSTPercentage = 21 Or pdblSTPercentage = 22 Then
      
         If pdblSTPercentage = 20 Then
            varSTAmount_Reg = Val(Format((varSTAmount / pdblSTPercentage) * 17, "####0.00"))
            varSTAmount_Ext = varSTAmount - varSTAmount_Reg
         Else
            varSTAmount_Reg = Val(Format((varSTAmount / pdblSTPercentage) * 18, "####0.00"))
            varSTAmount_Ext = varSTAmount - varSTAmount_Reg
         End If
         
         fpgTransaction.SetText 32, plngRowNo, Format(varSTAmount_Reg, "##,##0.00")
         fpgTransaction.SetText 33, plngRowNo, Format(varSTAmount_Ext, "##,##0.00")
         
      Else
      
         fpgTransaction.SetText 32, plngRowNo, varSTAmount
      
      End If
      
   Else
      'S.T VALUE
      fpgTransaction.SetText 12, plngRowNo, 0
      
   End If
   
   'SET THE INC. S.E.D AND S.T VALUE AGAIN
   fpgTransaction.SetText 10, plngRowNo, Format(Val(Format(varExcSEDnSTValue, "####0.00")) + _
                                                Val(Format(varSTAmount, "####0.00")) + _
                                                Val(Format(varSEDAmount, "####0.00")), "##,##0.00")
   
   'DISCOUNT 1 - PERCENTAGE
   fpgTransaction.GetText 27, plngRowNo, varDisc1_Percentage
   fpgTransaction.SetText 28, plngRowNo, Format(varExcSEDnSTValue * Val(varDisc1_Percentage) / 100, "##,##0.00")
   
   'DISCOUNT 2 - PER PACK
   fpgTransaction.GetText 29, plngRowNo, varDisc2_PerPack
   fpgTransaction.SetText 30, plngRowNo, Format(varNoOfPacks * Val(varDisc2_PerPack), "##,##0.00")
   
   'EMPTY CARTONS
   fpgTransaction.GetText 19, plngRowNo, varEmptyCartonPercentage
   fpgTransaction.SetText 15, plngRowNo, Format(Val(varNoOfPacks) * (Val(varEmptyCartonPercentage) / 100), "##,##0")
End Sub

Private Sub PrintCommercialInvoice(prptCommInvReport As arSalInvCommercialInvoice, pstrOCCopy As String, _
                                   pblnSendToPrinter As Boolean, pblnHideCoName As Boolean, Optional pblnIsExport As Boolean)
   Dim rsTemp As New ADODB.Recordset
   
   Dim strTotalPacks As String
   
   strTotalPacks = ", (SELECT SUM(NoOfPacks) FROM " & mstrSalesTableName & "Detail SEDtl WHERE SEDtl.CompanyID = " & Val(gstrOrgID) & " AND " & _
                      "SEDtl.FinYear = SE.FinYear AND SEDtl.SalesEntryType = SE.SalesEntryType AND SEDtl.SalesEntryID =  SE.SalesEntryID) AS TotalPacks"

   rsTemp.Open "SELECT SE.FinYear, SE.SalesEntryType, SE.SalesEntryID, SalesEntryDate, SE.CustomerID, CustomerCoName, Address, FG.Brand_ID, " & _
               "Brand_Title, SE_Dtl.FG_ID, FGName, NoOfPacks, NetRate, IncSTValue, TotalIncSTValue, NetAmount, TotalWeightInTons, VehicleRegNo, " & _
               "TotalDiscount1_Percentage_Value, TotalDiscount2_PerPack_Value, TotalCommission, SE.IncomeTaxPercentage, IncomeTax, " & _
               "IncomeTaxIncValue, (TotalDiscount1_Percentage_Value + TotalDiscount2_PerPack_Value) AS TotalDiscount, TotalExcSTValue, " & _
               "BasicRate, ExcSTValue, VATValue " & _
               strTotalPacks & " FROM " & mstrSalesTableName & " SE INNER JOIN " & mstrSalesTableName & "Detail SE_Dtl ON SE.CompanyID = " & _
               "SE_Dtl.CompanyID AND SE.FinYear = SE_Dtl.FinYear AND SE.SalesEntryType = SE_Dtl.SalesEntryType AND SE.SalesEntryID = " & _
               "SE_Dtl.SalesEntryID AND SE.BillType = SE_Dtl.BillType INNER JOIN tblSalCustomerInfo Cust ON SE.CompanyID = Cust.CompanyID AND " & _
               "SE.CustomerID = Cust.CustomerID INNER JOIN tblInvItemFG FG ON SE_Dtl.CompanyID = FG.CompanyID AND SE_Dtl.FG_ID = FG.FG_ID " & _
               "INNER JOIN tblInvItemFGBrand FGBrand ON FG.Brand_ID = FGBrand.Brand_ID WHERE SE.CompanyID = " & Val(gstrOrgID) & " AND " & _
               "SE.FinYear = '" & gstrFinancialYear & "' AND SE.BillType = 'SI' AND SE.SalesEntryType = 'BG' AND " & _
               "SE.SalesEntryID = " & mebSalesEntryID.text & " ORDER BY SE.SalesEntryType, SE.SalesEntryID, Brand_Title, SE_Dtl.FG_ID", _
               AppCnn, adOpenKeyset, adLockReadOnly

   If rsTemp.BOF = True And rsTemp.EOF = True Then
      MsgBox "Commercial Invoice - " & pstrOCCopy & " cannot be printed.", vbExclamation + vbOKOnly, App.ProductName
      Exit Sub
   End If

   With prptCommInvReport

      .dc.Recordset = rsTemp

      .Toolbar.Tools(2).Visible = mUserPriv.CanPrint

      If mUserPriv.CanExport = True Then
         .Toolbar.Tools.Add "Export"
         .Toolbar.Tools.Item(.Toolbar.Tools.Count - 1).ID = 786
         .Toolbar.Tools.Item(.Toolbar.Tools.Count - 1).Type = 0
      End If

      .dc.ConnectionString = AppCnn.ConnectionString
      .dc.UserID = gstrdbUser
      .dc.Password = gstrdbUserPassword
      
      If pblnHideCoName = True Then
         .lblOrgName.Visible = False
      End If
      
      .Caption = "Sales Invoice"
      
      .GroupHeader1.DataField = "SalesEntryID"
      
      .GroupHeader2.DataField = "Brand_Title"
      .fldBrandName1.DataField = "Brand_Title"
      .fldBrandName2.DataField = "Brand_Title"

      .fldInvoiceType.DataField = "SalesEntryType"
      .fldInvoiceNo.DataField = "SalesEntryID"
      .fldInvoiceDate.DataField = "SalesEntryDate"
      .fldCustomerCode.DataField = "CustomerID"
      .fldCustomerName.DataField = "CustomerCoName"
      .fldAddress.DataField = "Address"

      .fldFG_ID.DataField = "FG_ID"
      .fldProduct.DataField = "FGName"
      
      .fldNoOfPacks.DataField = "NoOfPacks"
      .fldPrice.DataField = "BasicRate"
      .fldExcValue.DataField = "ExcSTValue"
      .fldVAT.DataField = "VATValue"
      .fldNetPrice.DataField = "NetRate"
      .fldIncValue.DataField = "IncSTValue"

      .fldGrandTotalIncValue.DataField = "TotalIncSTValue"
      
      .fldTotalPacks.DataField = "TotalPacks"
      
      .GroupFooter1.Height = 2520
      
      Dim strSIDate As String
      
      strSIDate = rsTemp.Fields("SalesEntryDate").Value
      
      If pblnIsExport = True Then
         .Run
         
         Call ExportReport("PDF", "BG-" & Right("0000" & mebSalesEntryID.text, 5) & "-" & Replace(strSIDate, "/", "."), prptCommInvReport)
                           
         Call Unload(prptCommInvReport)
      
      Else
         If pblnSendToPrinter = False Then
            .Show
            
         Else
            .PrintReport False
            Unload prptCommInvReport
            
         End If
         
      End If
      
   End With
End Sub
